        function handleAnswer(value) {
            const question = questions[currentQuestion];
            const actualValue = question.reverse ? 100 - value : value;
            answers[question.id] = actualValue;

            if (currentQuestion < questions.length - 1) {
                currentQuestion++;
                render();
            } else {
                showResults = true;
                render();
            }
        }

        function calculateResults() {
            const personalQuestions = questions.filter(q => q.category === "personal");
            const studyQuestions = questions.filter(q => q.category === "study");
            const colleagueQuestions = questions.filter(q => q.category === "colleague");

            const personalScore = personalQuestions.reduce((sum, q) => sum + (answers[q.id] || 0), 0) / personalQuestions.length;
            const studyScore = studyQuestions.reduce((sum, q) => sum + (answers[q.id] || 0), 0) / studyQuestions.length;
            const colleagueScore = colleagueQuestions.reduce((sum, q) => sum + (answers[q.id] || 0), 0) / colleagueQuestions.length;
            const totalScore = (personalScore + studyScore + colleagueScore) / 3;

            return { personalScore, studyScore, colleagueScore, totalScore };
        }

        function getLevel(score) {
            if (score < 33) return { levelAr: "منخفض", levelEn: "Low", color: "green" };
            if (score < 66) return { levelAr: "متوسط", levelEn: "Moderate", color: "yellow" };
            return { levelAr: "مرتفع", levelEn: "High", color: "red" };
        }
